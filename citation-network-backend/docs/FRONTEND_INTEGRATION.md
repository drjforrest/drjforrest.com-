ResearchAPI {  private baseUrl: string;  constructor(baseUrl: string = API_BASE_URL) {    this.baseUrl = baseUrl;  }  async getPapers(): Promise<ResearchNetworkData> {    const response = await fetch(`${this.baseUrl}/api/papers`);    if (!response.ok) {      throw new Error('Failed to fetch papers');    }    return response.json();  }  async getTimeline(): Promise<TimelineData> {    const response = await fetch(`${this.baseUrl}/api/papers/timeline`);    if (!response.ok) {      throw new Error('Failed to fetch timeline');    }    return response.json();  }  async findSimilar(abstract: string): Promise<SimilarityResult> {    const response = await fetch(`${this.baseUrl}/api/papers/similar`, {      method: 'POST',      headers: {        'Content-Type': 'application/json',      },      body: JSON.stringify({ abstract }),    });        if (!response.ok) {      throw new Error('Failed to find similar papers');    }    return response.json();  }  async healthCheck(): Promise<HealthStatus> {    const response = await fetch(`${this.baseUrl}/api/health`);    if (!response.ok) {      throw new Error('API health check failed');    }    return response.json();  }}export const researchAPI = new ResearchAPI();```---## Custom Hook```typescript// src/hooks/useResearchNetwork.tsimport { useState, useEffect } from 'react';import { researchAPI } from '@/utils/api';import type { ResearchNetworkData, Paper } from '@/types/research';export function useResearchNetwork() {  const [data, setData] = useState<ResearchNetworkData | null>(null);  const [loading, setLoading] = useState(true);  const [error, setError] = useState<Error | null>(null);  const [selectedPaper, setSelectedPaper] = useState<Paper | null>(null);  const [yearRange, setYearRange] = useState<[number, number] | null>(null);  useEffect(() => {    async function fetchData() {      try {        setLoading(true);        const papers = await researchAPI.getPapers();        setData(papers);                // Calculate year range        if (papers.papers.length > 0) {          const years = papers.papers.map(p => p.year).filter(Boolean);          setYearRange([Math.min(...years), Math.max(...years)]);        }      } catch (err) {        setError(err as Error);      } finally {        setLoading(false);      }    }    fetchData();  }, []);  const filteredPapers = data?.papers.filter(paper => {    if (!yearRange) return true;    return paper.year >= yearRange[0] && paper.year <= yearRange[1];  });  return {    data,    filteredPapers,    loading,    error,    selectedPaper,    setSelectedPaper,    yearRange,    setYearRange,  };}```---## Main Network Graph Component```typescript// src/components/ResearchNetwork/NetworkGraph.tsximport { useEffect, useRef } from 'react';import * as d3 from 'd3';import type { Paper, Cluster } from '@/types/research';interface NetworkGraphProps {  papers: Paper[];  clusters: Cluster[];  onPaperClick: (paper: Paper) => void;}export function NetworkGraph({ papers, clusters, onPaperClick }: NetworkGraphProps) {  const svgRef = useRef<SVGSVGElement>(null);  useEffect(() => {    if (!svgRef.current || papers.length === 0) return;    const svg = d3.select(svgRef.current);    const width = svgRef.current.clientWidth;    const height = svgRef.current.clientHeight;    // Clear existing content    svg.selectAll('*').remove();    // Create color scale    const colorScale = d3.scaleOrdinal<number, string>()      .domain(clusters.map(c => c.id))      .range(clusters.map(c => c.color));    // Create zoom behavior    const zoom = d3.zoom<SVGSVGElement, unknown>()      .scaleExtent([0.5, 5])      .on('zoom', (event) => {        g.attr('transform', event.transform);      });    svg.call(zoom);    // Main group for zoom    const g = svg.append('g');    // Scale positions to fit viewport    const xExtent = d3.extent(papers, d => d.position.x) as [number, number];    const yExtent = d3.extent(papers, d => d.position.y) as [number, number];    const xScale = d3.scaleLinear()      .domain(xExtent)      .range([100, width - 100]);    const yScale = d3.scaleLinear()      .domain(yExtent)      .range([100, height - 100]);    // Draw connections (optional - connect papers in same cluster)    // ... (add force simulation or manual links if desired)    // Draw nodes    const nodes = g.selectAll('circle')      .data(papers)      .join('circle')      .attr('cx', d => xScale(d.position.x))      .attr('cy', d => yScale(d.position.y))      .attr('r', d => Math.sqrt(d.citations) + 5)      .attr('fill', d => colorScale(d.cluster))      .attr('opacity', 0.7)      .attr('stroke', '#fff')      .attr('stroke-width', 2)      .style('cursor', 'pointer')      .on('click', (event, d) => {        onPaperClick(d);      })      .on('mouseover', function(event, d) {        d3.select(this)          .transition()          .duration(200)          .attr('opacity', 1)          .attr('r', Math.sqrt(d.citations) + 8);      })      .on('mouseout', function(event, d) {        d3.select(this)          .transition()          .duration(200)          .attr('opacity', 0.7)          .attr('r', Math.sqrt(d.citations) + 5);      });    // Add labels for highly cited papers    const topPapers = papers      .sort((a, b) => b.citations - a.citations)      .slice(0, 10);    g.selectAll('text')      .data(topPapers)      .join('text')      .attr('x', d => xScale(d.position.x))      .attr('y', d => yScale(d.position.y) - Math.sqrt(d.citations) - 10)      .attr('text-anchor', 'middle')      .attr('font-size', '10px')      .attr('fill', '#333')      .text(d => d.title.split(':')[0].substring(0, 30) + '...');  }, [papers, clusters, onPaperClick]);  return (    <svg      ref={svgRef}      className="w-full h-full"      style={{ minHeight: '600px' }}    />  );}```---## Environment Variables```bash# .env.local (Vercel)NEXT_PUBLIC_API_URL=https://api.drjforrest.com# For development#
